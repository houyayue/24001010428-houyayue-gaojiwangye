核心代码

// ① 拼接口地址（limit 由输入框决定）
const url = 'https://dummyjson.com/products?limit=' + limit;

// ② fetch 是异步的，用 await 等到响应
const res = await fetch(url);
if (!res.ok) throw new Error('HTTP ' + res.status);   // ③ 404/500 要自己判断

// ④ 解析 JSON，拿到商品数组
const data = await res.json();

// ⑤ 遍历数组，拼 HTML 一次性插入（比循环 appendChild 更高效）
let html = '';
data.products.forEach(function (p) {
  html += '<div>'
        +   '<img src="' + p.thumbnail + '">'
        +   '<h3>' + p.title + '</h3>'
        +   '<b>￥' + p.price + '</b>'
        + '</div>';
});
grid.innerHTML = html;
